create table aircraft_types
(
    id           int auto_increment
        primary key,
    type_code    varchar(20)                                 not null,
    manufacturer varchar(50)                                 null,
    model        varchar(50)                                 null,
    seat_layout  json                                        null,
    status       enum ('active', 'retired') default 'active' null,
    constraint uk_type_code
        unique (type_code)
)
    comment '机型管理';

create table flights
(
    id               int auto_increment
        primary key,
    flight_no        varchar(20)                                                                                             not null,
    aircraft_type_id int                                                                                                     not null,
    origin_airport   varchar(10)                                                                                             not null,
    dest_airport     varchar(10)                                                                                             not null,
    sched_dep_time   datetime                                                                                                not null,
    sched_arr_time   datetime                                                                                                not null,
    status           enum ('scheduled', 'delayed', 'cancelled', 'boarding', 'departed', 'arrived') default 'scheduled'       null,
    route_info       json                                                                                                    null,
    created_at       datetime                                                                      default CURRENT_TIMESTAMP null,
    updated_at       datetime                                                                      default CURRENT_TIMESTAMP null on update CURRENT_TIMESTAMP,
    constraint uk_flight_no_time
        unique (flight_no, sched_dep_time),
    constraint fk_flights_aircraft
        foreign key (aircraft_type_id) references aircraft_types (id)
)
    comment '航班信息';

create index idx_status
    on flights (status);

create table users
(
    id                int auto_increment
        primary key,
    username          varchar(50)                           not null,
    password          varchar(255)                          not null,
    id_card           varchar(20)                           null,
    phone             varchar(15)                           null,
    real_name         varchar(50)                           null,
    role              varchar(20) default 'passenger'       not null,
    status            varchar(20) default 'active'          not null,
    registration_time datetime    default CURRENT_TIMESTAMP null,
    updated_time      datetime    default CURRENT_TIMESTAMP null on update CURRENT_TIMESTAMP,
    constraint id_card
        unique (id_card),
    constraint username
        unique (username)
)
    comment '用户信息表';

create table frequent_passengers
(
    id           bigint auto_increment comment '主键ID'
        primary key,
    user_id      int                                                  not null comment '所属用户ID（关联users表，实现数据隔离）',
    name         varchar(50)                                          not null comment '乘客姓名',
    id_card      varchar(18)                                          not null comment '身份证号（18位）',
    relationship varchar(20)                                          null comment '与用户的关系（配偶、子女、父母、兄弟姐妹、朋友、同事、其他）',
    phone        varchar(15)                                          null comment '联系电话',
    remarks      text                                                 null comment '备注信息',
    is_default   tinyint(1)                 default 0                 null comment '是否设为默认（0-否，1-是）',
    status       enum ('active', 'deleted') default 'active'          null comment '状态（active-有效，deleted-已删除）',
    created_at   datetime                   default CURRENT_TIMESTAMP null comment '创建时间',
    updated_at   datetime                   default CURRENT_TIMESTAMP null on update CURRENT_TIMESTAMP comment '更新时间',
    constraint uk_user_id_card
        unique (user_id, id_card) comment '同一用户下身份证号唯一',
    constraint fk_frequent_passenger_user
        foreign key (user_id) references users (id)
            on update cascade on delete cascade
)
    comment '常用乘客信息表（每个用户的数据完全隔离）' collate = utf8mb4_unicode_ci;

create index idx_id_card
    on frequent_passengers (id_card)
    comment '身份证号索引（用于验证唯一性）';

create index idx_user_id
    on frequent_passengers (user_id)
    comment '用户ID索引（用于快速查询该用户的常用乘客）';

create index idx_user_status
    on frequent_passengers (user_id, status)
    comment '用户ID和状态联合索引';

create table orders
(
    id             bigint auto_increment
        primary key,
    passenger_id   int                                                                                     not null comment '乘客ID（关联users表，实现数据隔离）',
    order_no       varchar(40)                                                                             not null comment '订单号',
    passenger_name varchar(50)                                                                             null comment '姓名（冗余字段，便于查询，从users.real_name同步）',
    route          varchar(100)                                                                            null comment '航线（格式：出发地 → 目的地，如：北京 → 上海）',
    departure_time datetime                                                                                null comment '起飞时间',
    total_amount   decimal(10, 2)                                                                          not null comment '订单总金额',
    status         enum ('created', 'paid', 'ticketed', 'cancelled', 'refunded') default 'created'         null comment '订单状态',
    created_at     datetime                                                      default CURRENT_TIMESTAMP null comment '创建时间',
    updated_at     datetime                                                      default CURRENT_TIMESTAMP null on update CURRENT_TIMESTAMP comment '更新时间',
    constraint uk_order_no
        unique (order_no),
    constraint fk_order_user
        foreign key (passenger_id) references users (id)
            on update cascade on delete cascade
)
    comment '订单主表（乘客隔离，包含客户姓名、航线、起飞时间）' collate = utf8mb4_unicode_ci;

create index idx_created_at
    on orders (created_at);

create index idx_departure_time
    on orders (departure_time);

create index idx_passenger
    on orders (passenger_id);

create index idx_passenger_name
    on orders (passenger_name);

create index idx_route
    on orders (route);

create index idx_status
    on orders (status);
